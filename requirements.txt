# Hiatal Tension Pipeline 3.0.0: Python dependencies
#
# Needs Python 3.9 or newer (tested on 3.9, 3.10, 3.12 and 3.14). Install into a virtual environment:
#     python3 -m venv .venv
#     source .venv/bin/activate          (Windows: .venv\Scripts\activate)
#     python3 -m pip install --upgrade pip      (needed on Python 3.9: its own pip is too old for current wheels)
#     pip install -r requirements.txt
#
# Versions are the oldest that were tested. The comment on each line says which scripts need the package.
# The exact versions of a passing environment are in requirements-tested.txt (use it to reproduce an environment).
# No separate FFmpeg install is needed: PyAV bundles its own FFmpeg libraries.

# --- Core: the main stages (index_video, extract_point_forces, plan_measurements, mark_suture_entry,
#     merge_distance_force) -------------------------------------------------------------------------
numpy>=2.0        # arrays and statistics: every stage
openpyxl>=3.1     # reads and writes the Excel workbooks: extract_point_forces, plan_measurements, merge_distance_force, make_tracking_sheet
matplotlib>=3.9   # the interactive windows: mark_suture_entry, calibrate_reference, pick_calibration_frame, measure_suture

# --- Video: reads the original video exactly, frame by frame ------------------------------------------
av>=15           # PyAV (bundles FFmpeg libraries): index_video, mark_suture_entry, export_calibration_frame, pick_calibration_frame

# --- Calibration frame export ----------------------------------------------------------------------------
pillow>=11.3      # lossless PNG of an exact frame: export_calibration_frame, pick_calibration_frame (matplotlib already needs it)

# --- Calibration and image tools -----------------------------------------------------------------------
opencv-python>=5.0  # image reading and feature matching: calibrate_reference, camera_check, measure_suture, add_timestamp, verify_force

# --- Force log parsing ---------------------------------------------------------------------------------
pandas>=2.3       # parse_force_api (and verify_force)

# Optional and macOS only (verify_force.py): see requirements-optional-macos.txt
#
# The interactive tools open a window. macOS and Windows normally need nothing more. On Linux install
# Tk (for example `sudo apt install python3-tk`) or `pip install PyQt6`, and if matplotlib picks no
# window backend set MPLBACKEND (for example MPLBACKEND=TkAgg or MPLBACKEND=QtAgg).
